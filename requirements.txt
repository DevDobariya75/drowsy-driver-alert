# Runtime. Target Pi OS: Raspberry Pi OS (Legacy, 64-bit) = Debian 12 Bookworm, Python 3.11, aarch64.
#   * laptop (Windows, Python 3.11)               -> MediaPipe 0.10.21, NumPy 1.26
#   * Raspberry Pi OS Legacy 64-bit (Bookworm)    -> MediaPipe 0.10.18, NumPy 1.26   <- our Pi
#   * Raspberry Pi OS Trixie (Python 3.13)        -> MediaPipe 1.1.1,   NumPy 2.x    (also works)
# pip picks the right line by itself. Why 0.10.18 on the Pi: MediaPipe 0.10.20 and newer 0.10.x
# have NO Linux ARM64 (aarch64) wheel, so "mediapipe==0.10.21" cannot install on a Pi.
# 0.10.18 is the newest 0.10.x with one; it gives the same landmarks as 0.10.21 (checked on a recording).
# Every package below (and every dependency) installs on the Pi as a ready-made wheel, nothing compiles.
# A 32-bit Pi OS will NOT work (no MediaPipe for 32-bit ARM).
mediapipe==0.10.21; python_version < "3.13" and platform_machine != "aarch64"
mediapipe==0.10.18; python_version < "3.13" and platform_machine == "aarch64"
mediapipe==1.1.1; python_version >= "3.13"
numpy==1.26.4; python_version < "3.13"
numpy>=2.1,<3; python_version >= "3.13"
# MediaPipe 0.10.x needs jax; pinned so pip doesn't try newer jax that wants NumPy 2
jax==0.7.1; python_version < "3.13"
jaxlib==0.7.1; python_version < "3.13"
opencv-python==4.11.0.86
opencv-contrib-python==4.11.0.86   # MediaPipe needs it; pinned to the same version as opencv-python
PyYAML==6.0.3
psutil==7.2.2

# Tests
pytest==9.1.1

# Face landmark model: models/face_landmarker.task (in the repository, ~3.7 MB, no download needed)
